    if (e.altKey && (e.code === 'KeyI' || e.key === 'i' || e.key === 'I' || e.key === 'ш' || e.key === 'Ш')) {
      e.preventDefault();
      e.stopPropagation();
      window.toggleDevSwitcher();
    } else if (e.key === 'Escape') {
      const modal = getModal();
      if (modal && !modal.classList.contains('hidden')) {

          <span class="px-2 py-0.5 rounded text-[9px] font-black uppercase tracking-wider ${roleBadgeColor} shrink-0 ml-2">
            ${u.role}
          </span>
        </a>
      `;
    }).join('');
  }

  document.querySelectorAll('.dev-role-filter').forEach(btn => {
    btn.addEventListener('click', function() {
      document.querySelectorAll('.dev-role-filter').forEach(b => {
        b.classList.remove('bg-indigo-600', 'text-white', 'border-indigo-500');
        b.classList.add('bg-slate-800', 'text-slate-300', 'border-slate-700');
      });
      this.classList.remove('bg-slate-800', 'text-slate-300', 'border-slate-700');
      this.classList.add('bg-indigo-600', 'text-white', 'border-indigo-500');
      
      const role = this.getAttribute('data-filter');
      renderUsers(role);
    });
  });

from itertools import product
def F(x, y, z, w):
    # Выражение: ((x -> y) == (w -> z)) or (x and w)
    return ((not(x <= y)) == (w <= z)) or (x and w)
print("x y z w | F")
print("-" * 15)
for x, y, z, w in product([0, 1], repeat=4):
    res = F(x, y, z, w)
    if not res:
        print(f"{x} {y} {z} {w} | {int(res)}")
